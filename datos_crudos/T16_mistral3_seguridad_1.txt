import os
import jwt
import time
from typing import Optional, Dict, Any
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.hazmat.backends import default_backend
from dataclasses import dataclass

@dataclass
class JWTConfig:
    """Configuration for JWT operations with security best practices."""
    algorithm: str = "RS256"  # Asymmetric algorithm for better security
    access_token_expiry: int = 900  # 15 minutes in seconds
    refresh_token_expiry: int = 86400  # 24 hours in seconds
    issuer: str = "secure_app"
    audience: str = "user_auth"
    private_key_path: str = "jwt_private_key.pem"
    public_key_path: str = "jwt_public_key.pem"
    key_size: int = 4096  # RSA key size for strong security

class JWTManager:
    """Secure JWT token manager with key rotation support."""

    def __init__(self, config: Optional[JWTConfig] = None):
        self.config = config if config else JWTConfig()
        self._ensure_keys_exist()

    def _ensure_keys_exist(self) -> None:
        """Generate RSA key pair if they don't exist."""
        if not (os.path.exists(self.config.private_key_path) and
                os.path.exists(self.config.public_key_path)):
            self._generate_key_pair()

    def _generate_key_pair(self) -> None:
        """Generate and save RSA key pair with secure permissions."""
        private_key = rsa.generate_private_key(
            public_exponent=65537,
            key_size=self.config.key_size,
            backend=default_backend()
        )
        public_key = private_key.public_key()

        # Save private key with restricted permissions
        with open(self.config.private_key_path, "wb") as f:
            f.write(private_key.private_bytes(
                encoding=serialization.Encoding.PEM,
                format=serialization.PrivateFormat.PKCS8,
                encryption_algorithm=serialization.NoEncryption()
            ))
        os.chmod(self.config.private_key_path, 0o600)

        # Save public key
        with open(self.config.public_key_path, "wb") as f:
            f.write(public_key.public_bytes(
                encoding=serialization.Encoding.PEM,
                format=serialization.PublicFormat.SubjectPublicKeyInfo
            ))

    def _get_private_key(self) -> rsa.RSAPrivateKey:
        """Load private key from file."""
        with open(self.config.private_key_path, "rb") as f:
            return serialization.load_pem_private_key(
                f.read(),
                password=None,
                backend=default_backend()
            )

    def _get_public_key(self) -> rsa.RSAPublicKey:
        """Load public key from file."""
        with open(self.config.public_key_path, "rb") as f:
            return serialization.load_pem_public_key(
                f.read(),
                backend=default_backend()
            )

    def generate_tokens(self, user_id: str, additional_claims: Optional[Dict[str, Any]] = None) -> Dict[str, str]:
        """
        Generate access and refresh tokens with security best practices.

        Args:
            user_id: Unique identifier for the user
            additional_claims: Optional additional claims to include in the token

        Returns:
            Dictionary containing access_token and refresh_token
        """
        if additional_claims is None:
            additional_claims = {}

        now = int(time.time())
        access_token_expiry = now + self.config.access_token_expiry
        refresh_token_expiry = now + self.config.refresh_token_expiry

        # Common claims for both tokens
        common_claims = {
            "iss": self.config.issuer,
            "aud": self.config.audience,
            "iat": now,
            "nbf": now,  # Not valid before
            "jti": os.urandom(16).hex(),  # Unique token identifier
            "sub": str(user_id),  # Subject (user ID)
            **additional_claims
        }

        # Generate access token
        access_token = jwt.encode(
            {**common_claims, "exp": access_token_expiry, "token_type": "access"},
            self._get_private_key(),
            algorithm=self.config.algorithm
        )

        # Generate refresh token
        refresh_token = jwt.encode(
            {**common_claims, "exp": refresh_token_expiry, "token_type": "refresh"},
            self._get_private_key(),
            algorithm=self.config.algorithm
        )

        return {
            "access_token": access_token,
            "refresh_token": refresh_token,
            "expires_in": self.config.access_token_expiry
        }

    def verify_token(self, token: str, expected_token_type: str = "access") -> Dict[str, Any]:
        """
        Verify and decode a JWT token with security checks.

        Args:
            token: JWT token to verify
            expected_token_type: Expected token type (access or refresh)

        Returns:
            Decoded token payload

        Raises:
            jwt.ExpiredSignatureError: If token has expired
            jwt.InvalidTokenError: If token is invalid for any reason
        """
        try:
            payload = jwt.decode(
                token,
                self._get_public_key(),
                algorithms=[self.config.algorithm],
                issuer=self.config.issuer,
                audience=self.config.audience,
                options={
                    "require": ["exp", "iat", "nbf", "iss", "aud", "sub", "jti", "token_type"],
                    "verify_exp": True,
                    "verify_iat": True,
                    "verify_nbf": True,
                    "verify_iss": True,
                    "verify_aud": True,
                }
            )

            # Additional security checks
            if payload["token_type"] != expected_token_type:
                raise jwt.InvalidTokenError("Invalid token type")

            # Check for token reuse (in a real app, you'd track jti in a database)
            if "jti" not in payload:
                raise jwt.InvalidTokenError("Missing jti claim")

            return payload

        except jwt.PyJWTError as e:
            # Log the error in a real application
            raise jwt.InvalidTokenError(f"Token verification failed: {str(e)}")

    def refresh_access_token(self, refresh_token: str) -> Dict[str, str]:
        """
        Refresh an access token using a valid refresh token.

        Args:
            refresh_token: Valid refresh token

        Returns:
            Dictionary containing new access_token and refresh_token

        Raises:
            jwt.InvalidTokenError: If refresh token is invalid
        """
        # Verify the refresh token first
        refresh_payload = self.verify_token(refresh_token, "refresh")

        # Generate new tokens
        return self.generate_tokens(
            refresh_payload["sub"],
            {k: v for k, v in refresh_payload.items() if k not in ["exp", "iat", "nbf", "jti", "token_type"]}
        )

# Example usage (would be in a separate file in a real application)
if __name__ == "__main__":
    # Initialize JWT manager
    jwt_manager = JWTManager()

    # Example user ID
    user_id = "user123"

    # Generate tokens
    tokens = jwt_manager.generate_tokens(user_id, {"role": "admin"})
    print("Generated tokens:")
    print(f"Access Token: {tokens['access_token'][:20]}...")
    print(f"Refresh Token: {tokens['refresh_token'][:20]}...")

    # Verify access token
    try:
        payload = jwt_manager.verify_token(tokens["access_token"])
        print("\nVerified access token payload:")
        print(payload)
    except jwt.InvalidTokenError as e:
        print(f"Token verification failed: {e}")

    # Refresh access token
    try:
        new_tokens = jwt_manager.refresh_access_token(tokens["refresh_token"])
        print("\nRefreshed tokens:")
        print(f"New Access Token: {new_tokens['access_token'][:20]}...")
    except jwt.InvalidTokenError as e:
        print(f"Token refresh failed: {e}")
