```python
import secrets
import hashlib
import time
from typing import Optional, Tuple

# Configuración
TOKEN_LENGTH = 32  # Longitud del token en bytes
TOKEN_EXPIRY = 3600  # Tiempo de expiración en segundos (1 hora)

def generate_password_reset_token(user_id: str) -> Tuple[str, str]:
    """
    Genera un token seguro para restablecer la contraseña de un usuario.

    Args:
        user_id: Identificador único del usuario.

    Returns:
        Tuple con el token en formato hexadecimal y el hash del token para almacenamiento.
    """
    # Generar un token aleatorio
    token = secrets.token_hex(TOKEN_LENGTH)

    # Crear un hash del token para almacenamiento seguro
    token_hash = hashlib.sha256(token.encode()).hexdigest()

    # Crear un payload con el user_id y timestamp de expiración
    expiry_timestamp = int(time.time()) + TOKEN_EXPIRY
    payload = f"{user_id}:{expiry_timestamp}"

    # Firmar el payload con el token hash
    signature = hashlib.sha256(f"{payload}:{token_hash}".encode()).hexdigest()

    # Devolver el token y el hash para almacenamiento
    return f"{payload}:{signature}", token_hash

def validate_password_reset_token(token: str, stored_token_hash: str) -> Optional[str]:
    """
    Valida un token de restablecimiento de contraseña.

    Args:
        token: Token recibido del usuario.
        stored_token_hash: Hash del token almacenado en la base de datos.

    Returns:
        El user_id si el token es válido, None en caso contrario.
    """
    try:
        # Dividir el token en sus componentes
        payload, signature = token.rsplit(':', 1)
        user_id, expiry_timestamp = payload.split(':')

        # Verificar que el token no haya expirado
        if int(time.time()) > int(expiry_timestamp):
            return None

        # Verificar la firma
        expected_signature = hashlib.sha256(f"{payload}:{stored_token_hash}".encode()).hexdigest()
        if not secrets.compare_digest(signature, expected_signature):
            return None

        # Verificar que el hash del token coincida con el almacenado
        token_hash = hashlib.sha256(token.encode()).hexdigest()
        if not secrets.compare_digest(token_hash, stored_token_hash):
            return None

        return user_id
    except (ValueError, AttributeError):
        return None
```
