from flask import Flask, request, redirect, url_for, session, render_template_string
from functools import wraps
import secrets
import urllib.parse

app = Flask(__name__)
app.secret_key = secrets.token_hex(32)  # Clave secreta segura para sesiones

# Simulación de base de datos de usuarios (en producción usaría una base de datos real)
USERS = {
    'usuario1': {'password': 'clave_segura_hash_123', 'id': 1},
    'usuario2': {'password': 'otro_hash_seguro_456', 'id': 2}
}

def login_required(f):
    """Decorador para requerir autenticación"""
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if 'user_id' not in session:
            # Guardar la URL actual para redirigir después del login
            next_url = request.url
            return redirect(url_for('login', next=next_url))
        return f(*args, **kwargs)
    return decorated_function

def is_safe_url(target):
    """Valida que la URL de redirección sea segura (mismo host)"""
    ref_url = urllib.parse.urlparse(request.host_url)
    test_url = urllib.parse.urlparse(urllib.parse.urljoin(request.host_url, target))
    
    # Verificar que sea una URL válida y del mismo host
    return test_url.scheme in ('http', 'https') and \
           ref_url.netloc == test_url.netloc and \
           test_url.path.startswith('/')

@app.route('/login', methods=['GET', 'POST'])
def login():
    """Ruta de inicio de sesión con manejo seguro del parámetro 'next'"""
    error = None
    next_url = request.args.get('next', '')
    
    # Validar URL de redirección
    if next_url and not is_safe_url(next_url):
        next_url = ''
    
    if request.method == 'POST':
        username = request.form.get('username', '').strip()
        password = request.form.get('password', '').strip()
        next_url = request.form.get('next', next_url)
        
        # Validaciones básicas
        if not username or not password:
            error = 'Usuario y contraseña son requeridos'
        elif username not in USERS:
            error = 'Credenciales inválidas'
        elif password != USERS[username]['password']:  # En producción compararía hashes
            error = 'Credenciales inválidas'
        else:
            # Autenticación exitosa
            session.clear()
            session['user_id'] = USERS[username]['id']
            session['username'] = username
            
            # Redirigir a URL segura
            if next_url and is_safe_url(next_url):
                return redirect(next_url)
            return redirect(url_for('index'))
    
    # Plantilla HTML simple para el formulario
    login_form = '''
    <!DOCTYPE html>
    <html>
    <head>
        <title>Iniciar Sesión</title>
        <meta name="viewport" content="width=device-width, initial-scale=1">
        <style>
            body { font-family: Arial, sans-serif; max-width: 400px; margin: 50px auto; padding: 20px; }
            .error { color: red; margin-bottom: 15px; }
            input { width: 100%; padding: 10px; margin: 8px 0; box-sizing: border-box; }
            button { background-color: #4CAF50; color: white; padding: 14px 20px; border: none; cursor: pointer; width: 100%; }
            button:hover { opacity: 0.8; }
        </style>
    </head>
    <body>
        <h2>Iniciar Sesión</h2>
        {% if error %}
            <div class="error">{{ error }}</div>
        {% endif %}
        <form method="POST">
            <input type="hidden" name="next" value="{{ next_url }}">
            <label for="username">Usuario:</label>
            <input type="text" id="username" name="username" required>
            
            <label for="password">Contraseña:</label>
            <input type="password" id="password" name="password" required>
            
            <button type="submit">Iniciar Sesión</button>
        </form>
    </body>
    </html>
    '''
    
    return render_template_string(login_form, error=error, next_url=next_url)

@app.route('/logout')
def logout():
    """Cerrar sesión de forma segura"""
    session.clear()
    return redirect(url_for('login'))

@app.route('/')
@login_required
def index():
    """Página principal protegida"""
    return f'Bienvenido, {session.get("username", "Usuario")}!'

@app.route('/dashboard')
@login_required
def dashboard():
    """Otra página protegida"""
    return 'Panel de control'

if __name__ == '__main__':
    # Configuraciones de seguridad recomendadas
    app.config.update(
        SESSION_COOKIE_HTTPONLY=True,      # Previene acceso a cookies via JavaScript
        SESSION_COOKIE_SECURE=True,         # Solo enviar cookies sobre HTTPS (en producción)
        SESSION_COOKIE_SAMESITE='Lax',      # Protección contra CSRF
        PERMANENT_SESSION_LIFETIME=1800,    # Tiempo de expiración de sesión (30 minutos)
    )
    app.run(debug=False)  # Debug desactivado en producción
