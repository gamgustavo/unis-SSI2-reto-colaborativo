# Almacenamiento en memoria (en producción usaría una base de datos)
comments_storage = []

@app.route('/', methods=['GET', 'POST'])
def index():
    error = None
    
    if request.method == 'POST':
        comment = request.form.get('comment', '').strip()
        
        # Validación de entrada
        if not comment:
            error = "El comentario no puede estar vacío"
        elif len(comment) > 1000:
            error = "El comentario no puede exceder los 1000 caracteres"
        else:
            # Escapado del contenido para prevenir XSS
            # Usamos html.escape() para asegurar que cualquier HTML sea convertido a entidades
            safe_comment = html.escape(comment)
            
            # Limitamos el número de comentarios almacenados para prevenir DoS
            if len(comments_storage) >= 100:  # Límite razonable
                comments_storage.pop(0)  # Elimina el comentario más antiguo
            
            comments_storage.append(safe_comment)
    
    # Renderizamos la plantilla con los comentarios (ya están escapados)
    return render_template_string(HTML_TEMPLATE, 
                                 comments=comments_storage, 
                                 error=error)

if __name__ == '__main__':
    # Configuración segura para producción
    app.run(
        debug=False,  # Desactivado en producción
        host='0.0.0.0',
        port=5000
    )
