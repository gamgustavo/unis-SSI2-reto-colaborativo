```python
import os
import secrets
import hashlib
import hmac
import time
from typing import Optional, Tuple
from dataclasses import dataclass

# Configuración segura
TOKEN_LENGTH = 32  # Longitud del token en bytes (256 bits)
TOKEN_EXPIRATION = 3600  # 1 hora en segundos
SECRET_KEY = os.getenv('RESET_TOKEN_SECRET_KEY')  # Debe ser una clave secreta fuerte y única

if not SECRET_KEY or len(SECRET_KEY) < 32:
    raise ValueError("RESET_TOKEN_SECRET_KEY debe ser una cadena aleatoria de al menos 32 caracteres")

@dataclass
class PasswordResetToken:
    token_hash: str
    user_id: str
    expires_at: int
    salt: str

def generate_password_reset_token(user_id: str) -> Tuple[str, PasswordResetToken]:
    """
    Genera un token seguro para restablecer la contraseña.

    Args:
        user_id: Identificador único del usuario

    Returns:
        Tuple con el token en texto claro (para enviar al usuario) y el objeto PasswordResetToken
    """
    if not user_id:
        raise ValueError("user_id no puede estar vacío")

    # Generar token aleatorio seguro
    token = secrets.token_hex(TOKEN_LENGTH)

    # Generar salt único para este token
    salt = secrets.token_hex(16)

    # Calcular hash del token con HMAC-SHA256
    token_hash = _hash_token(token, salt)

    # Calcular tiempo de expiración
    expires_at = int(time.time()) + TOKEN_EXPIRATION

    # Crear objeto con los datos del token
    token_data = PasswordResetToken(
        token_hash=token_hash,
        user_id=user_id,
        expires_at=expires_at,
        salt=salt
    )

    return token, token_data

def validate_password_reset_token(token: str, token_data: PasswordResetToken) -> bool:
    """
    Valida un token de restablecimiento de contraseña.

    Args:
        token: Token en texto claro proporcionado por el usuario
        token_data: Objeto PasswordResetToken almacenado en la base de datos

    Returns:
        True si el token es válido, False en caso contrario
    """
    if not token or not token_data:
        return False

    # Verificar expiración
    if time.time() > token_data.expires_at:
        return False

    # Verificar que el token coincide con el hash almacenado
    expected_hash = _hash_token(token, token_data.salt)
    return hmac.compare_digest(expected_hash, token_data.token_hash)

def _hash_token(token: str, salt: str) -> str:
    """
    Genera un hash seguro del token usando HMAC-SHA256.

    Args:
        token: Token en texto claro
        salt: Salt único para este token

    Returns:
        Hash hexadecimal del token
    """
    if not token or not salt:
        raise ValueError("Token y salt no pueden estar vacíos")

    # Usar HMAC con la clave secreta para mayor seguridad
    hmac_obj = hmac.new(
        SECRET_KEY.encode('utf-8'),
        msg=f"{token}{salt}".encode('utf-8'),
        digestmod=hashlib.sha256
    )
    return hmac_obj.hexdigest()

# Ejemplo de uso (no forma parte de la implementación real)
if __name__ == "__main__":
    # Configurar clave secreta para el ejemplo (en producción usa variables de entorno)
    os.environ['RESET_TOKEN_SECRET_KEY'] = secrets.token_hex(32)

    # Generar token
    user_id = "user123"
    token, token_data = generate_password_reset_token(user_id)
    print(f"Token generado: {token}")
    print(f"Datos del token: {token_data}")

    # Validar token (correcto)
    is_valid = validate_password_reset_token(token, token_data)
    print(f"Token válido: {is_valid}")

    # Validar token (incorrecto)
    is_valid = validate_password_reset_token("token_incorrecto", token_data)
    print(f"Token inválido: {is_valid}")

    # Validar token expirado (simular expiración)
    token_data.expires_at = int(time.time()) - 1
    is_valid = validate_password_reset_token(token, token_data)
    print(f"Token expirado: {is_valid}")
```
