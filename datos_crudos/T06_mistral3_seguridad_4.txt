```python
import os
import secrets
import hashlib
import hmac
import time
from typing import Optional, Tuple
from dataclasses import dataclass
import base64
import json

# Configuración segura (en producción, usa variables de entorno o un sistema de configuración seguro)
RESET_TOKEN_SECRET = os.getenv('RESET_TOKEN_SECRET', secrets.token_hex(32))
RESET_TOKEN_EXPIRATION = 3600  # 1 hora en segundos
RESET_TOKEN_SALT = b'password_reset_token_salt'

@dataclass
class PasswordResetToken:
    user_id: str
    expires_at: int
    token_hash: str

def _generate_secure_token() -> str:
    """Genera un token seguro criptográficamente aleatorio."""
    return secrets.token_urlsafe(32)

def _hash_token(token: str) -> str:
    """Genera un hash seguro del token usando HMAC-SHA256."""
    return hmac.new(
        RESET_TOKEN_SECRET.encode(),
        msg=token.encode(),
        digestmod=hashlib.sha256
    ).hexdigest()

def _serialize_token_data(token_data: PasswordResetToken) -> str:
    """Serializa los datos del token de forma segura."""
    data = {
        'user_id': token_data.user_id,
        'expires_at': token_data.expires_at,
        'token_hash': token_data.token_hash
    }
    return base64.urlsafe_b64encode(json.dumps(data).encode()).decode()

def _deserialize_token_data(serialized: str) -> Optional[PasswordResetToken]:
    """Deserializa los datos del token de forma segura."""
    try:
        data = json.loads(base64.urlsafe_b64decode(serialized.encode()).decode())
        return PasswordResetToken(
            user_id=data['user_id'],
            expires_at=data['expires_at'],
            token_hash=data['token_hash']
        )
    except (json.JSONDecodeError, KeyError, UnicodeDecodeError):
        return None

def generate_password_reset_token(user_id: str) -> Tuple[str, str]:
    """
    Genera un token seguro para restablecer la contraseña.

    Args:
        user_id: Identificador único del usuario.

    Returns:
        Tuple con el token en texto plano (para enviar al usuario) y el token serializado (para almacenar).
    """
    if not user_id:
        raise ValueError("User ID cannot be empty")

    token_plain = _generate_secure_token()
    expires_at = int(time.time()) + RESET_TOKEN_EXPIRATION
    token_hash = _hash_token(token_plain)

    token_data = PasswordResetToken(
        user_id=user_id,
        expires_at=expires_at,
        token_hash=token_hash
    )

    serialized_token = _serialize_token_data(token_data)
    return token_plain, serialized_token

def validate_password_reset_token(token_plain: str, serialized_token: str) -> Optional[str]:
    """
    Valida un token de restablecimiento de contraseña.

    Args:
        token_plain: Token en texto plano proporcionado por el usuario.
        serialized_token: Token serializado almacenado en el sistema.

    Returns:
        user_id si el token es válido, None en caso contrario.
    """
    if not token_plain or not serialized_token:
        return None

    token_data = _deserialize_token_data(serialized_token)
    if not token_data:
        return None

    # Verificar expiración
    if time.time() > token_data.expires_at:
        return None

    # Verificar hash del token
    expected_hash = _hash_token(token_plain)
    if not hmac.compare_digest(expected_hash, token_data.token_hash):
        return None

    return token_data.user_id

# Ejemplo de uso (no forma parte de la implementación real)
if __name__ == "__main__":
    # Generar token
    user_id = "user123"
    token_plain, serialized_token = generate_password_reset_token(user_id)
    print(f"Token generado para {user_id}:")
    print(f"Token (enviar al usuario): {token_plain}")
    print(f"Token serializado (almacenar): {serialized_token}")

    # Validar token
    valid_user_id = validate_password_reset_token(token_plain, serialized_token)
    print(f"\nValidación del token:")
    print(f"Token válido para usuario: {valid_user_id}")

    # Token inválido (expirado)
    time.sleep(2)
    expired_token = PasswordResetToken(
        user_id=user_id,
        expires_at=int(time.time()) - 10,
        token_hash=_hash_token(token_plain)
    )
    expired_serialized = _serialize_token_data(expired_token)
    print(f"Token expirado válido: {validate_password_reset_token(token_plain, expired_serialized) is None}")

    # Token inválido (hash incorrecto)
    invalid_token = PasswordResetToken(
        user_id=user_id,
        expires_at=int(time.time()) + 3600,
        token_hash="incorrect_hash"
    )
    invalid_serialized = _serialize_token_data(invalid_token)
    print(f"Token con hash incorrecto válido: {validate_password_reset_token(token_plain, invalid_serialized) is None}")
```
