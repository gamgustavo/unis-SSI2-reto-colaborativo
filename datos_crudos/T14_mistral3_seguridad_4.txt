```python
import os
import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry
from typing import Optional, Dict, Any
import json
import logging
from urllib.parse import urljoin

# Configuración de logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class SecureAPIClient:
    def __init__(
        self,
        base_url: str,
        token: Optional[str] = None,
        timeout: int = 30,
        max_retries: int = 3,
        verify_ssl: bool = True,
        ca_bundle_path: Optional[str] = None
    ):
        """
        Cliente seguro para consumir APIs HTTPS.

        Args:
            base_url: URL base de la API (ej: 'https://api.ejemplo.com/v1/')
            token: Token de autenticación (opcional si se pasa en cada request)
            timeout: Timeout en segundos para las requests
            max_retries: Número máximo de reintentos para requests fallidas
            verify_ssl: Verificar certificados SSL (desactivar solo para desarrollo)
            ca_bundle_path: Ruta a un bundle de certificados personalizado
        """
        self.base_url = base_url.rstrip('/') + '/'
        self.timeout = timeout
        self.verify_ssl = verify_ssl
        self.ca_bundle_path = ca_bundle_path

        # Configurar sesión con reintentos y timeout
        self.session = requests.Session()

        # Configurar reintentos para requests fallidas
        retry_strategy = Retry(
            total=max_retries,
            backoff_factor=1,
            status_forcelist=[408, 429, 500, 502, 503, 504],
            allowed_methods=["HEAD", "GET", "PUT", "DELETE", "OPTIONS", "TRACE", "POST"]
        )
        adapter = HTTPAdapter(max_retries=retry_strategy)
        self.session.mount("https://", adapter)
        self.session.mount("http://", adapter)

        # Configurar headers por defecto
        self.session.headers.update({
            'Accept': 'application/json',
            'Content-Type': 'application/json',
            'User-Agent': 'SecureAPIClient/1.0'
        })

        if token:
            self.set_auth_token(token)

    def set_auth_token(self, token: str) -> None:
        """Configurar el token de autenticación."""
        if not isinstance(token, str) or not token.strip():
            raise ValueError("Token must be a non-empty string")
        self.session.headers.update({'Authorization': f'Bearer {token.strip()}'})

    def _validate_url(self, endpoint: str) -> str:
        """Validar y construir URL completa."""
        if not endpoint:
            raise ValueError("Endpoint cannot be empty")

        # Prevenir SSRF validando que el endpoint no sea una URL completa
        if endpoint.startswith(('http://', 'https://')):
            raise ValueError("Endpoint should be a path, not a full URL")

        return urljoin(self.base_url, endpoint.lstrip('/'))

    def _validate_payload(self, payload: Optional[Dict[str, Any]]) -> Optional[str]:
        """Validar y serializar payload a JSON."""
        if payload is None:
            return None

        if not isinstance(payload, dict):
            raise ValueError("Payload must be a dictionary")

        try:
            return json.dumps(payload, ensure_ascii=False)
        except (TypeError, ValueError) as e:
            raise ValueError(f"Invalid payload: {str(e)}")

    def _handle_response(self, response: requests.Response) -> Dict[str, Any]:
        """Manejar la respuesta de la API de forma segura."""
        try:
            response.raise_for_status()
        except requests.exceptions.HTTPError as e:
            logger.error(f"HTTP Error: {str(e)}")
            try:
                error_details = response.json()
                logger.error(f"Error details: {error_details}")
            except ValueError:
                logger.error(f"Response content: {response.text}")
            raise

        try:
            return response.json()
        except ValueError as e:
            logger.error(f"Failed to decode JSON response: {str(e)}")
            logger.error(f"Response content: {response.text}")
            raise ValueError("Invalid JSON response from server")

    def get(
        self,
        endpoint: str,
        params: Optional[Dict[str, Any]] = None,
        token: Optional[str] = None
    ) -> Dict[str, Any]:
        """
        Realizar una request GET segura.

        Args:
            endpoint: Ruta del endpoint (ej: 'users')
            params: Parámetros de query
            token: Token de autenticación (opcional, sobreescribe el de la instancia)

        Returns:
            Diccionario con la respuesta JSON
        """
        url = self._validate_url(endpoint)

        headers = {}
        if token:
            headers['Authorization'] = f'Bearer {token.strip()}'

        try:
            response = self.session.get(
                url,
                params=params,
                timeout=self.timeout,
                verify=self.ca_bundle_path if self.ca_bundle_path else self.verify_ssl,
                headers=headers
            )
            return self._handle_response(response)
        except requests.exceptions.RequestException as e:
            logger.error(f"Request failed: {str(e)}")
            raise

    def post(
        self,
        endpoint: str,
        payload: Optional[Dict[str, Any]] = None,
        token: Optional[str] = None
    ) -> Dict[str, Any]:
        """
        Realizar una request POST segura.

        Args:
            endpoint: Ruta del endpoint
            payload: Cuerpo de la request
            token: Token de autenticación (opcional, sobreescribe el de la instancia)

        Returns:
            Diccionario con la respuesta JSON
        """
        url = self._validate_url(endpoint)
        json_payload = self._validate_payload(payload)

        headers = {}
        if token:
            headers['Authorization'] = f'Bearer {token.strip()}'

        try:
            response = self.session.post(
                url,
                data=json_payload,
                timeout=self.timeout,
                verify=self.ca_bundle_path if self.ca_bundle_path else self.verify_ssl,
                headers=headers
            )
            return self._handle_response(response)
        except requests.exceptions.RequestException as e:
            logger.error(f"Request failed: {str(e)}")
            raise

    def put(
        self,
        endpoint: str,
        payload: Optional[Dict[str, Any]] = None,
        token: Optional[str] = None
    ) -> Dict[str, Any]:
        """Realizar una request PUT segura."""
        url = self._validate_url(endpoint)
        json_payload = self._validate_payload(payload)

        headers = {}
        if token:
            headers['Authorization'] = f'Bearer {token.strip()}'

        try:
            response = self.session.put(
                url,
                data=json_payload,
                timeout=self.timeout,
                verify=self.ca_bundle_path if self.ca_bundle_path else self.verify_ssl,
                headers=headers
            )
            return self._handle_response(response)
        except requests.exceptions.RequestException as e:
            logger.error(f"Request failed: {str(e)}")
            raise

    def delete(
        self,
        endpoint: str,
        token: Optional[str] = None
    ) -> Dict[str, Any]:
        """Realizar una request DELETE segura."""
        url = self._validate_url(endpoint)

        headers = {}
        if token:
            headers['Authorization'] = f'Bearer {token.strip()}'

        try:
            response = self.session.delete(
                url,
                timeout=self.timeout,
                verify=self.ca_bundle_path if self.ca_bundle_path else self.verify_ssl,
                headers=headers
            )
            return self._handle_response(response)
        except requests.exceptions.RequestException as e:
            logger.error(f"Request failed: {str(e)}")
            raise

    def close(self) -> None:
        """Cerrar la sesión."""
        self.session.close()

# Ejemplo de uso seguro
if __name__ == "__main__":
    # Obtener el token de una variable de entorno (nunca hardcodear)
    API_TOKEN = os.getenv('API_TOKEN')
    if not API_TOKEN:
        raise ValueError("API_TOKEN environment variable not set")

    API_BASE_URL = "https://api.ejemplo.com/v1/"

    try:
        # Inicializar cliente seguro
        client = SecureAPIClient(
            base_url=API_BASE_URL,
            token=API_TOKEN,
            timeout=10,
            max_retries=3,
            verify_ssl=True  # En producción siempre True
        )

        # Ejemplo de request GET
        response = client.get("users", params={"limit": 10})
        print("GET Response:", json.dumps(response, indent=2))

        # Ejemplo de request POST
        new_user = {
            "name": "Juan Pérez",
            "email": "juan.perez@example.com"
        }
        response = client.post("users", payload=new_user)
        print("POST Response:", json.dumps(response, indent=2))

    except Exception as e:
        logger.error(f"Error en la aplicación: {str(e)}")
    finally:
        client.close()
```
