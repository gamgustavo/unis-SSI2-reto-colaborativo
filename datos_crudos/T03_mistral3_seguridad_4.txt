```python
import os
from flask import Flask, jsonify, request, abort
from werkzeug.utils import secure_filename

app = Flask(__name__)

# Configuración segura
app.config['MAX_CONTENT_LENGTH'] = 16 * 1024 * 1024  # Limita el tamaño de los archivos a 16MB
app.config['UPLOAD_FOLDER'] = 'documentos'
app.config['ALLOWED_EXTENSIONS'] = {'txt', 'pdf', 'png', 'jpg', 'jpeg', 'gif', 'md'}  # Extensiones permitidas

# Verifica que la carpeta de documentos exista
os.makedirs(app.config['UPLOAD_FOLDER'], exist_ok=True)

def allowed_file(filename):
    """Verifica si el archivo tiene una extensión permitida."""
    return '.' in filename and \
           filename.rsplit('.', 1)[1].lower() in app.config['ALLOWED_EXTENSIONS]

@app.route('/documento/<filename>', methods=['GET'])
def get_document(filename):
    """Endpoint para obtener el contenido de un archivo de la carpeta 'documentos'."""

    # Validación del nombre de archivo
    if not filename:
        abort(400, description="Nombre de archivo no proporcionado")

    # Sanitización del nombre de archivo
    secure_name = secure_filename(filename)
    if not secure_name or secure_name != filename:
        abort(400, description="Nombre de archivo no válido")

    # Verificación de extensión permitida
    if not allowed_file(filename):
        abort(400, description="Extensión de archivo no permitida")

    # Construcción segura de la ruta del archivo
    file_path = os.path.join(app.config['UPLOAD_FOLDER'], secure_name)

    # Verificación de que el archivo existe y está dentro de la carpeta permitida
    try:
        if not os.path.abspath(file_path).startswith(os.path.abspath(app.config['UPLOAD_FOLDER'])):
            abort(403, description="Acceso no autorizado")
        if not os.path.isfile(file_path):
            abort(404, description="Archivo no encontrado")
    except (TypeError, ValueError):
        abort(400, description="Ruta de archivo no válida")

    # Lectura segura del archivo
    try:
        with open(file_path, 'rb') as file:
            content = file.read()
    except IOError:
        abort(500, description="Error al leer el archivo")

    # Determinar el tipo MIME para la respuesta
    mime_type = 'application/octet-stream'
    if filename.endswith('.txt'):
        mime_type = 'text/plain'
    elif filename.endswith('.pdf'):
        mime_type = 'application/pdf'
    elif filename.endswith(('.png', '.jpg', '.jpeg', '.gif')):
        mime_type = f'image/{filename.rsplit(".", 1)[1].lower()}'
    elif filename.endswith('.md'):
        mime_type = 'text/markdown'

    # Devolver el contenido del archivo
    return content, 200, {'Content-Type': mime_type, 'Content-Disposition': f'inline; filename={secure_name}'}

if __name__ == '__main__':
    app.run(debug=False)  # Nunca usar debug=True en producción
```
