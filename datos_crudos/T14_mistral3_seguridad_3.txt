```python
import os
import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry
from typing import Optional, Dict, Any
import json
import logging
from urllib.parse import urljoin

# Configuración de logging para debugging (opcional)
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class SecureAPIClient:
    def __init__(
        self,
        base_url: str,
        token: str,
        timeout: int = 10,
        max_retries: int = 3,
        verify_ssl: bool = True,
        ca_bundle_path: Optional[str] = None
    ):
        """
        Inicializa el cliente seguro para consumir una API HTTPS.

        Args:
            base_url: URL base de la API (ej: 'https://api.ejemplo.com/v1/')
            token: Token de autenticación (se recomienda no hardcodearlo)
            timeout: Tiempo máximo de espera para la respuesta (segundos)
            max_retries: Número máximo de reintentos en caso de fallos
            verify_ssl: Verificar certificado SSL (recomendado True)
            ca_bundle_path: Ruta a un bundle de certificados personalizado (opcional)
        """
        if not base_url.startswith(('https://', 'http://')):
            raise ValueError("La URL base debe comenzar con http:// o https://")

        self.base_url = base_url.rstrip('/') + '/'
        self.token = token
        self.timeout = timeout
        self.verify_ssl = verify_ssl
        self.ca_bundle_path = ca_bundle_path

        # Configurar sesión con reintentos y timeout
        self.session = self._create_session(max_retries)

        # Validar SSL si está habilitado
        if self.verify_ssl:
            self._validate_ssl()

    def _create_session(self, max_retries: int) -> requests.Session:
        """Crea una sesión con configuración de reintentos y timeout."""
        session = requests.Session()

        # Configurar reintentos para errores temporales
        retry_strategy = Retry(
            total=max_retries,
            backoff_factor=1,
            status_forcelist=[408, 429, 500, 502, 503, 504],
            allowed_methods=["HEAD", "GET", "PUT", "DELETE", "OPTIONS", "TRACE", "POST"]
        )
        adapter = HTTPAdapter(max_retries=retry_strategy)
        session.mount("https://", adapter)
        session.mount("http://", adapter)

        return session

    def _validate_ssl(self) -> None:
        """Valida la configuración SSL antes de hacer requests."""
        if self.ca_bundle_path:
            if not os.path.isfile(self.ca_bundle_path):
                raise FileNotFoundError(f"El bundle de certificados no existe: {self.ca_bundle_path}")
            self.session.verify = self.ca_bundle_path
        else:
            # Usar certifi (incluido en requests) como bundle por defecto
            import certifi
            self.session.verify = certifi.where()

    def _get_headers(self) -> Dict[str, str]:
        """Genera headers seguros para la request."""
        return {
            'Authorization': f'Bearer {self.token}',
            'User-Agent': 'SecureAPIClient/1.0',
            'Accept': 'application/json',
            'Content-Type': 'application/json',
        }

    def _handle_response(self, response: requests.Response) -> Dict[str, Any]:
        """Maneja la respuesta de la API de forma segura."""
        try:
            response.raise_for_status()
            return response.json()
        except requests.exceptions.JSONDecodeError:
            logger.error("Respuesta no es JSON válido")
            raise ValueError("La respuesta de la API no es un JSON válido")
        except requests.exceptions.HTTPError as e:
            logger.error(f"Error HTTP: {e}")
            raise
        except requests.exceptions.RequestException as e:
            logger.error(f"Error en la request: {e}")
            raise

    def get(self, endpoint: str, params: Optional[Dict] = None) -> Dict[str, Any]:
        """
        Realiza una petición GET segura a la API.

        Args:
            endpoint: Ruta del endpoint (ej: 'users')
            params: Parámetros de query (opcional)

        Returns:
            Diccionario con la respuesta JSON
        """
        url = urljoin(self.base_url, endpoint)
        headers = self._get_headers()

        try:
            response = self.session.get(
                url,
                headers=headers,
                params=params,
                timeout=self.timeout,
                verify=self.verify_ssl
            )
            return self._handle_response(response)
        except requests.exceptions.RequestException as e:
            logger.error(f"Error en GET {url}: {e}")
            raise

    def post(self, endpoint: str, data: Optional[Dict] = None) -> Dict[str, Any]:
        """
        Realiza una petición POST segura a la API.

        Args:
            endpoint: Ruta del endpoint (ej: 'users')
            data: Datos a enviar en el body (opcional)

        Returns:
            Diccionario con la respuesta JSON
        """
        url = urljoin(self.base_url, endpoint)
        headers = self._get_headers()

        try:
            response = self.session.post(
                url,
                headers=headers,
                json=data,
                timeout=self.timeout,
                verify=self.verify_ssl
            )
            return self._handle_response(response)
        except requests.exceptions.RequestException as e:
            logger.error(f"Error en POST {url}: {e}")
            raise

    def put(self, endpoint: str, data: Optional[Dict] = None) -> Dict[str, Any]:
        """Realiza una petición PUT segura a la API."""
        url = urljoin(self.base_url, endpoint)
        headers = self._get_headers()

        try:
            response = self.session.put(
                url,
                headers=headers,
                json=data,
                timeout=self.timeout,
                verify=self.verify_ssl
            )
            return self._handle_response(response)
        except requests.exceptions.RequestException as e:
            logger.error(f"Error en PUT {url}: {e}")
            raise

    def delete(self, endpoint: str) -> Dict[str, Any]:
        """Realiza una petición DELETE segura a la API."""
        url = urljoin(self.base_url, endpoint)
        headers = self._get_headers()

        try:
            response = self.session.delete(
                url,
                headers=headers,
                timeout=self.timeout,
                verify=self.verify_ssl
            )
            return self._handle_response(response)
        except requests.exceptions.RequestException as e:
            logger.error(f"Error en DELETE {url}: {e}")
            raise

# Ejemplo de uso (comentar en producción)
if __name__ == "__main__":
    # Obtener el token de una variable de entorno (recomendado)
    API_TOKEN = os.getenv('API_TOKEN')
    if not API_TOKEN:
        raise ValueError("El token de API no está configurado en las variables de entorno")

    # Inicializar cliente
    client = SecureAPIClient(
        base_url="https://api.ejemplo.com/v1/",
        token=API_TOKEN,
        timeout=15,
        max_retries=3
    )

    try:
        # Ejemplo de petición GET
        response = client.get("users", params={"limit": 10})
        print("Respuesta:", json.dumps(response, indent=2))

        # Ejemplo de petición POST
        # response = client.post("users", data={"name": "Ejemplo"})
    except Exception as e:
        logger.error(f"Error en la operación: {e}")
```
