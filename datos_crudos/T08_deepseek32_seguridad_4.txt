@app.route('/', methods=['GET', 'POST'])
def index():
    comment = None
    
    if request.method == 'POST':
        # Obtener y limpiar el comentario
        user_comment = request.form.get('comment', '').strip()
        
        # Validación básica: longitud máxima razonable
        if len(user_comment) > 10000:
            user_comment = user_comment[:10000] + "... [comentario truncado]"
        
        # Escape seguro del comentario para prevenir XSS
        # Flask/Jinja2 ya hace escape automático, pero hacemos doble protección
        safe_comment = html.escape(user_comment)
        comment = safe_comment
    
    # Renderizar la plantilla con el comentario (si existe)
    # Jinja2 hará escape automático adicional
    return render_template_string(HTML_TEMPLATE, comment=comment)

if __name__ == '__main__':
    # Configuración de seguridad para producción
    app.config.update(
        SECRET_KEY='clave-secreta-compleja-y-larga-aqui',  # En producción, usar variable de entorno
        SESSION_COOKIE_HTTPONLY=True,
        SESSION_COOKIE_SECURE=False,  # True en producción con HTTPS
        SESSION_COOKIE_SAMESITE='Lax',
        PERMANENT_SESSION_LIFETIME=1800  # 30 minutos
    )
    
    # Solo para desarrollo
    app.run(debug=False, host='127.0.0.1', port=5000)
