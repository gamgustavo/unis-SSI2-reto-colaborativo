```python
import requests
import os

def get_api_data(api_url, api_token):
    headers = {
        'Authorization': f'Bearer {api_token}',
        'User-Agent':'my-app/1.0.0',  # Establecer un User-Agent puede ayudar a identificar tu aplicación en los logs del servidor
        'Accept': 'application/json'   # Especificar el tipo de contenido esperado
    }

    try:
        response = requests.get(api_url, headers=headers, timeout=10)  # Establecer un timeout para evitar esperas infinitas
        response.raise_for_status()  # Verificar si la respuesta tiene un código de error HTTP
        
        # Validar que la respuesta sea JSON
        response_json = response.json()
        
        return response_json
    
    except requests.exceptions.HTTPError as http_err:
        print(f'Error HTTP: {http_err}')
    except requests.exceptions.ConnectionError as conn_err:
        print(f'Error de Conexión: {conn_err}')
    except requests.exceptions.Timeout as timeout_err:
        print(f'Timeout: {timeout_err}')
    except requests.exceptions.RequestException as req_err:
        print(f'Error de Request: {req_err}')
    except ValueError:
        print('Respuesta no es un JSON válido')

# Obtener el token desde una variable de entorno para evitar exponer credenciales en el código
api_token = os.getenv('API_TOKEN')
api_url = 'https://api.example.com/data'

if api_token:
    data = get_api_data(api_url, api_token)
    if data:
        print(data)
else:
    print('El token de API no está definido.')
```
