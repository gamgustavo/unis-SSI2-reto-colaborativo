```python
import os
from flask import Flask, render_template_string
from werkzeug.middleware.proxy_fix import ProxyFix

app = Flask(__name__)

# Configuración segura por defecto
app.config.update(
    SECRET_KEY=os.urandom(32),  # Clave secreta aleatoria para sesiones
    SESSION_COOKIE_SECURE=True,  # Solo enviar cookies sobre HTTPS
    SESSION_COOKIE_HTTPONLY=True,  # Evitar acceso a cookies desde JavaScript
    SESSION_COOKIE_SAMESITE='Lax',  # Protección contra CSRF
    PERMANENT_SESSION_LIFETIME=1800,  # Sesiones expiran en 30 minutos
    PREFERRED_URL_SCHEME='https'  # Forzar HTTPS en URLs generadas
)

# Middleware para aplicaciones detrás de proxy (ej: Nginx)
app.wsgi_app = ProxyFix(app.wsgi_app, x_for=1, x_proto=1, x_host=1, x_prefix=1)

# Plantilla HTML segura con protección contra XSS
INDEX_TEMPLATE = """
<!DOCTYPE html>
<html lang="es">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Flask App Segura</title>
    <style>
        body { font-family: Arial, sans-serif; line-height: 1.6; max-width: 800px; margin: 0 auto; padding: 20px; }
        .security-info { background: #f8f9fa; padding: 15px; border-radius: 5px; margin-top: 20px; }
    </style>
</head>
<body>
    <h1>Bienvenido a la Aplicación Flask Segura</h1>
    <p>Esta es una implementación mínima con prácticas de seguridad básicas.</p>

    <div class="security-info">
        <h3>Características de seguridad implementadas:</h3>
        <ul>
            <li>Clave secreta aleatoria para sesiones</li>
            <li>Cookies seguras (Secure, HttpOnly, SameSite)</li>
            <li>Protección contra XSS en plantillas</li>
            <li>Configuración para HTTPS</li>
            <li>Soporte para proxy inverso</li>
            <li>Cabeceras de seguridad HTTP</li>
        </ul>
    </div>
</body>
</html>
"""

@app.route('/')
def index():
    """Ruta de inicio segura con plantilla protegida contra XSS"""
    return render_template_string(INDEX_TEMPLATE)

@app.after_request
def add_security_headers(response):
    """Añadir cabeceras de seguridad HTTP a todas las respuestas"""
    # Protección contra XSS y clickjacking
    response.headers['X-Content-Type-Options'] = 'nosniff'
    response.headers['X-Frame-Options'] = 'SAMEORIGIN'
    response.headers['X-XSS-Protection'] = '1; mode=block'

    # Política de seguridad de contenido (CSP)
    csp = (
        "default-src 'self'; "
        "script-src 'self'; "
        "style-src 'self' 'unsafe-inline'; "
        "img-src 'self' data:; "
        "font-src 'self'; "
        "object-src 'none'; "
        "base-uri 'self'; "
        "form-action 'self'; "
        "frame-ancestors 'self'"
    )
    response.headers['Content-Security-Policy'] = csp

    # Referrer Policy
    response.headers['Referrer-Policy'] = 'strict-origin-when-cross-origin'

    # Permisos de la API
    response.headers['Permissions-Policy'] = (
        "geolocation=(), microphone=(), camera=(), "
        "payment=(), usb=(), vr=()"
    )

    return response

if __name__ == '__main__':
    # Configuración para producción (usar servidor WSGI en producción)
    app.run(
        host='0.0.0.0',
        port=int(os.environ.get('PORT', 5000)),
        ssl_context='adhoc' if os.environ.get('FLASK_ENV') == 'development' else None,
        debug=False  # Nunca activar debug en producción
    )
```
