```python
import sqlite3
import hashlib
import secrets

def crear_tabla_usuarios():
    """Crea la tabla de usuarios si no existe"""
    conn = sqlite3.connect('usuarios.db')
    cursor = conn.cursor()
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS usuarios (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT UNIQUE NOT NULL,
            password_hash TEXT NOT NULL,
            salt TEXT NOT NULL
        )
    ''')
    conn.commit()
    conn.close()

def hash_password(password, salt=None):
    """Genera un hash seguro de la contraseña usando salt"""
    if salt is None:
        salt = secrets.token_hex(16)
    
    # Combinar contraseña con salt
    salted_password = password + salt
    
    # Crear hash SHA-256
    hash_obj = hashlib.sha256(salted_password.encode())
    password_hash = hash_obj.hexdigest()
    
    return password_hash, salt

def registrar_usuario(username, password):
    """Registra un nuevo usuario en la base de datos"""
    try:
        # Verificar que el nombre de usuario no esté vacío
        if not username or not username.strip():
            return False, "El nombre de usuario no puede estar vacío"
        
        # Verificar que la contraseña no esté vacía
        if not password:
            return False, "La contraseña no puede estar vacía"
        
        # Generar hash de la contraseña
        password_hash, salt = hash_password(password)
        
        # Guardar en la base de datos
        conn = sqlite3.connect('usuarios.db')
        cursor = conn.cursor()
        
        cursor.execute(
            "INSERT INTO usuarios (username, password_hash, salt) VALUES (?, ?, ?)",
            (username.strip(), password_hash, salt)
        )
        
        conn.commit()
        conn.close()
        
        return True, "Usuario registrado exitosamente"
    
    except sqlite3.IntegrityError:
        return False, "El nombre de usuario ya existe"
    except Exception as e:
        return False, f"Error al registrar usuario: {str(e)}"

def verificar_usuario(username, password):
    """Verifica las credenciales de un usuario"""
    try:
        conn = sqlite3.connect('usuarios.db')
        cursor = conn.cursor()
        
        # Buscar usuario en la base de datos
        cursor.execute(
            "SELECT password_hash, salt FROM usuarios WHERE username = ?",
            (username,)
        )
        
        resultado = cursor.fetchone()
        conn.close()
        
        if resultado is None:
            return False, "Usuario no encontrado"
        
        stored_hash, salt = resultado
        
        # Calcular hash de la contraseña proporcionada con el salt almacenado
        provided_hash, _ = hash_password(password, salt)
        
        # Comparar hashes de manera segura
        if secrets.compare_digest(provided_hash, stored_hash):
            return True, "Credenciales válidas"
        else:
            return False, "Contraseña incorrecta"
    
    except Exception as e:
        return False, f"Error al verificar usuario: {str(e)}"

# Crear la tabla al importar el módulo
crear_tabla_usuarios()

# Ejemplo de uso (comentado)
if __name__ == "__main__":
    # Registrar un usuario
    resultado, mensaje = registrar_usuario("usuario_ejemplo", "mi_contraseña_segura")
    print(f"Registro: {mensaje}")
    
    # Verificar credenciales correctas
    resultado, mensaje = verificar_usuario("usuario_ejemplo", "mi_contraseña_segura")
    print(f"Verificación correcta: {mensaje}")
    
    # Verificar credenciales incorrectas
    resultado, mensaje = verificar_usuario("usuario_ejemplo", "contraseña_incorrecta")
    print(f"Verificación incorrecta: {mensaje}")
```
