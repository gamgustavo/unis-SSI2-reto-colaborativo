```python
import os
import psycopg2
from psycopg2 import sql
from psycopg2.extras import RealDictCursor
from dotenv import load_dotenv
import logging
from typing import List, Dict, Optional

# Configurar logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# Cargar variables de entorno desde un archivo .env
load_dotenv()

class PostgreSQLClient:
    def __init__(self):
        self.connection = None
        self.cursor = None
        self._validate_environment_variables()

    def _validate_environment_variables(self) -> None:
        """Validar que las variables de entorno necesarias estén configuradas."""
        required_vars = [
            'DB_HOST', 'DB_PORT', 'DB_NAME',
            'DB_USER', 'DB_PASSWORD'
        ]
        missing_vars = [var for var in required_vars if not os.getenv(var)]
        if missing_vars:
            raise EnvironmentError(
                f"Faltan variables de entorno requeridas: {', '.join(missing_vars)}"
            )

    def _get_connection_params(self) -> Dict[str, str]:
        """Obtener parámetros de conexión desde variables de entorno."""
        return {
            'host': os.getenv('DB_HOST'),
            'port': os.getenv('DB_PORT'),
            'dbname': os.getenv('DB_NAME'),
            'user': os.getenv('DB_USER'),
            'password': os.getenv('DB_PASSWORD'),
            'sslmode': os.getenv('DB_SSL_MODE', 'require'),  # SSL por defecto
            'connect_timeout': int(os.getenv('DB_CONNECT_TIMEOUT', '5')),
            'application_name': os.getenv('DB_APPLICATION_NAME', 'clientes_app')
        }

    def connect(self) -> None:
        """Establecer conexión segura a la base de datos."""
        try:
            params = self._get_connection_params()
            self.connection = psycopg2.connect(**params)
            self.connection.autocommit = False  # Manejar transacciones explícitamente
            logger.info("Conexión a PostgreSQL establecida correctamente")
        except psycopg2.Error as e:
            logger.error(f"Error al conectar a PostgreSQL: {e}")
            raise

    def disconnect(self) -> None:
        """Cerrar la conexión de manera segura."""
        if self.cursor:
            self.cursor.close()
        if self.connection:
            self.connection.close()
            logger.info("Conexión a PostgreSQL cerrada")

    def execute_query(
        self,
        query: str,
        params: Optional[tuple] = None,
        fetch: bool = True
    ) -> Optional[List[Dict]]:
        """
        Ejecutar una consulta SQL de manera segura con parámetros.

        Args:
            query: Consulta SQL con placeholders (%s)
            params: Tupla de parámetros para la consulta
            fetch: Si True, devuelve los resultados

        Returns:
            Lista de diccionarios con los resultados o None
        """
        if not self.connection or self.connection.closed:
            self.connect()

        try:
            with self.connection.cursor(cursor_factory=RealDictCursor) as cursor:
                # Usar sql.SQL para componer consultas de manera segura
                if params:
                    cursor.execute(query, params)
                else:
                    cursor.execute(query)

                if fetch:
                    results = cursor.fetchall()
                    logger.info(f"Consulta ejecutada. Filas devueltas: {len(results)}")
                    return results
                else:
                    self.connection.commit()
                    logger.info("Consulta ejecutada sin devolver resultados")
                    return None

        except psycopg2.Error as e:
            self.connection.rollback()
            logger.error(f"Error al ejecutar consulta: {e}")
            raise

    def get_clientes(self, limit: int = 100) -> List[Dict]:
        """
        Obtener una lista de clientes de la tabla clientes.

        Args:
            limit: Número máximo de filas a devolver

        Returns:
            Lista de diccionarios con los datos de los clientes
        """
        try:
            # Usar sql.SQL para componer la consulta de manera segura
            query = sql.SQL("""
                SELECT id, nombre, email, telefono, direccion, fecha_registro
                FROM clientes
                WHERE activo = TRUE
                ORDER BY fecha_registro DESC
                LIMIT %s
            """)
            return self.execute_query(query, (limit,))
        except Exception as e:
            logger.error(f"Error al obtener clientes: {e}")
            raise

def main():
    try:
        # Inicializar cliente de base de datos
        db_client = PostgreSQLClient()
        db_client.connect()

        # Obtener y mostrar clientes
        clientes = db_client.get_clientes(limit=50)

        if clientes:
            print("\nListado de clientes:")
            print("-" * 80)
            for cliente in clientes:
                print(f"ID: {cliente['id']}")
                print(f"Nombre: {cliente['nombre']}")
                print(f"Email: {cliente['email']}")
                print(f"Teléfono: {cliente.get('telefono', 'N/A')}")
                print(f"Dirección: {cliente.get('direccion', 'N/A')}")
                print(f"Fecha registro: {cliente['fecha_registro']}")
                print("-" * 80)
        else:
            print("No se encontraron clientes activos.")

    except Exception as e:
        logger.error(f"Error en la aplicación: {e}")
    finally:
        if 'db_client' in locals():
            db_client.disconnect()

if __name__ == "__main__":
    main()
```
