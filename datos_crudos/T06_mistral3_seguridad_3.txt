```python
import os
import secrets
import hashlib
import hmac
import time
from typing import Optional, Tuple
from dataclasses import dataclass

# Configuración segura (en producción, usa variables de entorno o un sistema de configuración seguro)
RESET_TOKEN_SECRET = os.getenv('RESET_TOKEN_SECRET', secrets.token_hex(32))
RESET_TOKEN_EXPIRATION = 3600  # 1 hora en segundos
TOKEN_LENGTH = 32  # Longitud del token en bytes (antes de codificación)

@dataclass
class PasswordResetToken:
    token_hash: str
    user_id: str
    expires_at: int
    salt: str

def _generate_secure_token() -> str:
    """Genera un token criptográficamente seguro."""
    return secrets.token_hex(TOKEN_LENGTH)

def _hash_token(token: str, salt: str) -> str:
    """Genera un hash seguro del token usando HMAC-SHA256."""
    return hmac.new(
        key=RESET_TOKEN_SECRET.encode('utf-8'),
        msg=f"{token}{salt}".encode('utf-8'),
        digestmod=hashlib.sha256
    ).hexdigest()

def generate_password_reset_token(user_id: str) -> Tuple[str, PasswordResetToken]:
    """
    Genera un token seguro para restablecer contraseña.

    Args:
        user_id: Identificador único del usuario

    Returns:
        Tuple con el token en texto plano (para enviar al usuario) y el objeto PasswordResetToken
    """
    if not user_id:
        raise ValueError("user_id no puede estar vacío")

    token_plain = _generate_secure_token()
    salt = secrets.token_hex(16)
    expires_at = int(time.time()) + RESET_TOKEN_EXPIRATION

    token_hash = _hash_token(token_plain, salt)

    return token_plain, PasswordResetToken(
        token_hash=token_hash,
        user_id=user_id,
        expires_at=expires_at,
        salt=salt
    )

def validate_password_reset_token(
    token_plain: str,
    stored_token: PasswordResetToken,
    max_token_age: Optional[int] = None
) -> bool:
    """
    Valida un token de restablecimiento de contraseña.

    Args:
        token_plain: Token en texto plano proporcionado por el usuario
        stored_token: Objeto PasswordResetToken almacenado en la base de datos
        max_token_age: Edad máxima permitida para el token en segundos (opcional)

    Returns:
        bool: True si el token es válido, False en caso contrario
    """
    if not token_plain or not stored_token:
        return False

    # Verificar expiración
    current_time = int(time.time())
    if stored_token.expires_at < current_time:
        return False

    # Verificar edad máxima si se proporciona
    if max_token_age is not None:
        token_age = current_time - (stored_token.expires_at - RESET_TOKEN_EXPIRATION)
        if token_age > max_token_age:
            return False

    # Verificar hash del token
    computed_hash = _hash_token(token_plain, stored_token.salt)
    return hmac.compare_digest(computed_hash, stored_token.token_hash)

# Ejemplo de uso:
if __name__ == "__main__":
    # Generar token
    user_id = "user123"
    token_plain, token_data = generate_password_reset_token(user_id)
    print(f"Token generado (para enviar al usuario): {token_plain}")
    print(f"Datos del token (para almacenar en DB): {token_data}")

    # Validar token (simulando que viene del usuario)
    is_valid = validate_password_reset_token(token_plain, token_data)
    print(f"Token válido: {is_valid}")

    # Simular token expirado
    expired_token = PasswordResetToken(
        token_hash=token_data.token_hash,
        user_id=token_data.user_id,
        expires_at=int(time.time()) - 100,  # Ya expiró
        salt=token_data.salt
    )
    is_valid = validate_password_reset_token(token_plain, expired_token)
    print(f"Token expirado válido: {is_valid}")
```
