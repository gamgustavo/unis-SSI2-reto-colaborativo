```python
import tempfile
import os

def guardar_reporte_temporal(datos_reporte):
    try:
        # Crear un archivo temporal
        with tempfile.NamedTemporaryFile(delete=False, mode='w+', suffix='.txt') as temp_file:
            temp_file.write(datos_reporte)
            temp_file_path = temp_file.name
        
        print(f"Reporte guardado temporalmente en: {temp_file_path}")
        return temp_file_path
    
    except Exception as e:
        print(f"Error al guardar el reporte: {e}")
        return None

# Ejemplo de uso
if __name__ == "__main__":
    datos_ejemplo = "Este es un reporte de ejemplo."
    ruta_archivo = guardar_reporte_temporal(datos_ejemplo)
    if ruta_archivo:
        print(f"Ruta del archivo temporal: {ruta_archivo}")
        # No olvides borrar el archivo temporal cuando ya no lo necesites
        os.remove(ruta_archivo)
```
