import secrets
import string
import hashlib
import hmac
import time
from typing import Optional, Tuple
from dataclasses import dataclass
from datetime import datetime, timedelta
import json
import base64

@dataclass
class TokenValidationResult:
    is_valid: bool
    user_id: Optional[str] = None
    error: Optional[str] = None

class PasswordResetTokenManager:
    def __init__(self, secret_key: str, token_expiry_hours: int = 24):
        """
        Inicializa el gestor de tokens de restablecimiento de contraseña.
        
        Args:
            secret_key: Clave secreta para firmar tokens (debe ser larga y aleatoria)
            token_expiry_hours: Horas de validez del token (por defecto 24)
        """
        if not secret_key or len(secret_key) < 32:
            raise ValueError("La clave secreta debe tener al menos 32 caracteres")
        
        self.secret_key = secret_key.encode('utf-8')
        self.token_expiry_hours = token_expiry_hours
        
    def _generate_random_string(self, length: int = 32) -> str:
        """Genera una cadena aleatoria segura."""
        alphabet = string.ascii_letters + string.digits
        return ''.join(secrets.choice(alphabet) for _ in range(length))
    
    def _create_signature(self, data: str) -> str:
        """Crea un HMAC-SHA256 signature para los datos."""
        return hmac.new(
            self.secret_key,
            data.encode('utf-8'),
            hashlib.sha256
        ).hexdigest()
    
    def _encode_token_data(self, user_id: str, expiry_timestamp: int, nonce: str) -> str:
        """Codifica los datos del token en formato JSON base64 seguro."""
        token_data = {
            'user_id': user_id,
            'expiry': expiry_timestamp,
            'nonce': nonce,
            'version': '1.0'
        }
        
        # Convertir a JSON y luego a base64
        json_data = json.dumps(token_data, separators=(',', ':'))  # Compact JSON
        encoded_data = base64.urlsafe_b64encode(json_data.encode('utf-8')).decode('utf-8')
        
        return encoded_data
    
    def _decode_token_data(self, encoded_data: str) -> Optional[dict]:
        """Decodifica los datos del token desde base64."""
        try:
            # Añadir padding si es necesario
            padding_needed = 4 - len(encoded_data) % 4
            if padding_needed != 4:
                encoded_data += '=' * padding_needed
            
            json_data = base64.urlsafe_b64decode(encoded_data.encode('utf-8')).decode('utf-8')
            token_data = json.loads(json_data)
            
            # Validar estructura básica
            required_fields = {'user_id', 'expiry', 'nonce', 'version'}
            if not all(field in token_data for field in required_fields):
                return None
                
            return token_data
        except (ValueError, json.JSONDecodeError, UnicodeDecodeError):
            return None
    
    def generate_reset_token(self, user_id: str) -> Tuple[str, datetime]:
        """
        Genera un token seguro para restablecer la contraseña.
        
        Args:
            user_id: Identificador único del usuario
            
        Returns:
            Tuple con (token, fecha_de_expiracion)
        """
        # Validar entrada
        if not user_id or not isinstance(user_id, str):
            raise ValueError("user_id debe ser una cadena no vacía")
        
        # Generar componentes del token
        nonce = self._generate_random_string(16)
        expiry_timestamp = int(time.time()) + (self.token_expiry_hours * 3600)
        
        # Codificar datos
        encoded_data = self._encode_token_data(user_id, expiry_timestamp, nonce)
        
        # Crear firma
        signature = self._create_signature(encoded_data)
        
        # Combinar datos y firma
        token = f"{encoded_data}.{signature}"
        
        expiry_date = datetime.fromtimestamp(expiry_timestamp)
        
        return token, expiry_date
    
    def validate_reset_token(self, token: str) -> TokenValidationResult:
        """
        Valida un token de restablecimiento de contraseña.
        
        Args:
            token: Token a validar
            
        Returns:
            TokenValidationResult con el resultado de la validación
        """
        # Validar formato básico
        if not token or '.' not in token:
            return TokenValidationResult(
                is_valid=False,
                error="Formato de token inválido"
            )
        
        try:
            # Separar datos y firma
            encoded_data, signature = token.rsplit('.', 1)
            
            # Verificar firma primero (timing-safe)
            expected_signature = self._create_signature(encoded_data)
            if not hmac.compare_digest(signature, expected_signature):
                return TokenValidationResult(
                    is_valid=False,
                    error="Firma inválida"
                )
            
            # Decodificar datos
            token_data = self._decode_token_data(encoded_data)
            if not token_data:
                return TokenValidationResult(
                    is_valid=False,
                    error="Datos del token corruptos"
                )
            
            # Verificar versión
            if token_data.get('version') != '1.0':
                return TokenValidationResult(
                    is_valid=False,
                    error="Versión de token no soportada"
                )
            
            # Verificar expiración
            current_time = int(time.time())
            if current_time > token_data['expiry']:
                return TokenValidationResult(
                    is_valid=False,
                    error="Token expirado"
                )
            
            # Validar user_id
            user_id = token_data['user_id']
            if not user_id or not isinstance(user_id, str):
                return TokenValidationResult(
                    is_valid=False,
                    error="Identificador de usuario inválido"
                )
            
            # Validar nonce
            nonce = token_data['nonce']
            if not nonce or len(nonce) != 16:
                return TokenValidationResult(
                    is_valid=False,
                    error="Nonce inválido"
                )
            
            return TokenValidationResult(
                is_valid=True,
                user_id=user_id
            )
            
        except (ValueError, KeyError, TypeError) as e:
            return TokenValidationResult(
                is_valid=False,
                error=f"Error procesando token: {str(e)}"
            )
    
    def invalidate_token(self, token: str) -> bool:
        """
        Marca un token como utilizado/inválido.
        En una implementación real, esto registraría el token en una lista negra.
        
        Args:
            token: Token a invalidar
            
        Returns:
            True si se invalidó correctamente
        """
        # En una implementación completa, aquí se almacenaría el token
        # en una base de datos o caché como utilizado
        validation_result = self.validate_reset_token(token)
        
        if validation_result.is_valid:
            # Registrar token como utilizado (implementación específica)
            # Ejemplo: self._blacklist_token(token, validation_result.user_id)
            return True
        
        return False


# Ejemplo de uso
if __name__ == "__main__":
    # Configuración (en producción, usar una clave segura desde variables de entorno)
    SECRET_KEY = secrets.token_urlsafe(64)  # 64 bytes = 512 bits
    TOKEN_EXPIRY_HOURS = 24
    
    # Inicializar gestor de tokens
    token_manager = PasswordResetTokenManager(SECRET_KEY, TOKEN_EXPIRY_HOURS)
    
    # Ejemplo: Generar token para un usuario
    user_id = "usuario_123"
    token, expiry = token_manager.generate_reset_token(user_id)
    print(f"Token generado: {token[:50]}...")
    print(f"Expira: {expiry}")
    
    # Ejemplo: Validar token
    validation_result = token_manager.validate_reset_token(token)
    if validation_result.is_valid:
        print(f"Token válido para usuario: {validation_result.user_id}")
    else:
        print(f"Token inválido: {validation_result.error}")
    
    # Ejemplo: Token inválido
    invalid_token = "token.malformado"
    invalid_result = token_manager.validate_reset_token(invalid_token)
    print(f"Token inválido detectado: {invalid_result.error}")
