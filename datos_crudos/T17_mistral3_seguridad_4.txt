```python
import re
import ast
import operator
import math

# Diccionario seguro de funciones y constantes permitidas
SAFE_GLOBALS = {
    '__builtins__': None,
    'abs': abs,
    'min': min,
    'max': max,
    'pow': pow,
    'round': round,
    'sum': sum,
    'math': math,
}

# Operadores permitidos
SAFE_OPERATORS = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.Mod: operator.mod,
    ast.Pow: operator.pow,
    ast.USub: operator.neg,
    ast.BitXor: operator.xor,
    ast.FloorDiv: operator.floordiv,
}

def safe_eval(node):
    """Evalúa de forma segura un AST (Abstract Syntax Tree)"""
    if isinstance(node, ast.Num):  # Número
        return node.n
    elif isinstance(node, ast.BinOp):  # Operación binaria (ej: a + b)
        left = safe_eval(node.left)
        right = safe_eval(node.right)
        return SAFE_OPERATORS[type(node.op)](left, right)
    elif isinstance(node, ast.UnaryOp):  # Operación unaria (ej: -a)
        operand = safe_eval(node.operand)
        return SAFE_OPERATORS[type(node.op)](operand)
    elif isinstance(node, ast.Name):  # Variables/constantes permitidas
        if node.id in SAFE_GLOBALS:
            return SAFE_GLOBALS[node.id]
        raise ValueError(f"Nombre no permitido: {node.id}")
    elif isinstance(node, ast.Call):  # Llamadas a funciones permitidas
        if not isinstance(node.func, ast.Name):
            raise ValueError("Llamada a función no permitida")
        if node.func.id not in SAFE_GLOBALS:
            raise ValueError(f"Función no permitida: {node.func.id}")
        args = [safe_eval(arg) for arg in node.args]
        return SAFE_GLOBALS[node.func.id](*args)
    elif isinstance(node, ast.Attribute):  # Atributos de módulos permitidos (ej: math.pi)
        if not isinstance(node.value, ast.Name) or node.value.id != 'math':
            raise ValueError("Atributo no permitido")
        if not hasattr(math, node.attr):
            raise ValueError(f"Atributo no permitido: {node.attr}")
        return getattr(math, node.attr)
    else:
        raise ValueError(f"Expresión no permitida: {type(node).__name__}")

def validate_expression(expr):
    """Valida que la expresión solo contenga caracteres permitidos"""
    if not re.match(r'^[\d+\-*/()., \w]+$', expr):
        raise ValueError("Caracteres no permitidos en la expresión")
    return expr

def calculate(expr):
    """Calcula el resultado de una expresión matemática de forma segura"""
    try:
        expr = validate_expression(expr)
        node = ast.parse(expr, mode='eval').body
        return safe_eval(node)
    except (ValueError, SyntaxError, TypeError, ZeroDivisionError) as e:
        raise ValueError(f"Error en la expresión: {str(e)}")

def main():
    print("Calculadora segura. Escribe 'salir' para terminar.")
    while True:
        try:
            expr = input("> ").strip()
            if expr.lower() in ('salir', 'exit', 'quit'):
                break
            if not expr:
                continue
            result = calculate(expr)
            print(f"Resultado: {result}")
        except ValueError as e:
            print(f"Error: {str(e)}")
        except KeyboardInterrupt:
            print("\nSaliendo...")
            break
        except Exception as e:
            print(f"Error inesperado: {str(e)}")

if __name__ == "__main__":
    main()
```
