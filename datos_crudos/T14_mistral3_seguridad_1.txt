```python
import os
import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry
from typing import Optional, Dict, Any
import json
import logging
from urllib.parse import urljoin

# Configuración de logging
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class SecureAPIClient:
    def __init__(
        self,
        base_url: str,
        token: Optional[str] = None,
        timeout: int = 30,
        max_retries: int = 3,
        verify_ssl: bool = True,
        ca_bundle_path: Optional[str] = None
    ):
        """
        Inicializa el cliente API con configuración segura.

        Args:
            base_url: URL base de la API (ej: 'https://api.ejemplo.com/v1/')
            token: Token de autenticación (puede ser None si se configura después)
            timeout: Timeout en segundos para las peticiones
            max_retries: Número máximo de reintentos para peticiones fallidas
            verify_ssl: Verificar certificados SSL (True por defecto)
            ca_bundle_path: Ruta a un bundle de CA personalizado (opcional)
        """
        if not base_url.startswith(('https://', 'http://')):
            raise ValueError("La URL base debe comenzar con http:// o https://")

        self.base_url = base_url.rstrip('/') + '/'
        self.token = token
        self.timeout = timeout
        self.verify_ssl = verify_ssl
        self.ca_bundle_path = ca_bundle_path

        # Configurar sesión con reintentos y timeout
        self.session = self._create_session(max_retries)

        # Validar SSL si está habilitado
        if self.verify_ssl:
            self._validate_ssl_config()

    def _create_session(self, max_retries: int) -> requests.Session:
        """Crea una sesión con configuración de reintentos y timeout."""
        session = requests.Session()

        # Configurar reintentos para errores temporales
        retry_strategy = Retry(
            total=max_retries,
            backoff_factor=1,
            status_forcelist=[408, 429, 500, 502, 503, 504],
            allowed_methods=["HEAD", "GET", "PUT", "DELETE", "OPTIONS", "TRACE", "POST"]
        )

        adapter = HTTPAdapter(max_retries=retry_strategy)
        session.mount("https://", adapter)
        session.mount("http://", adapter)

        return session

    def _validate_ssl_config(self) -> None:
        """Valida la configuración SSL."""
        if not self.verify_ssl:
            logger.warning("La verificación SSL está deshabilitada. Esto es inseguro en producción.")
            return

        if self.ca_bundle_path:
            if not os.path.exists(self.ca_bundle_path):
                raise FileNotFoundError(f"El bundle de CA no existe: {self.ca_bundle_path}")
            if not os.access(self.ca_bundle_path, os.R_OK):
                raise PermissionError(f"No se puede leer el bundle de CA: {self.ca_bundle_path}")

    def _get_headers(self) -> Dict[str, str]:
        """Genera los headers para la petición."""
        headers = {
            "Accept": "application/json",
            "User-Agent": "SecureAPIClient/1.0",
            "Content-Type": "application/json"
        }

        if self.token:
            headers["Authorization"] = f"Bearer {self.token}"

        return headers

    def _handle_response(self, response: requests.Response) -> Dict[str, Any]:
        """Maneja la respuesta de la API de forma segura."""
        try:
            response.raise_for_status()
        except requests.exceptions.HTTPError as e:
            logger.error(f"Error HTTP {response.status_code}: {response.text}")
            raise

        try:
            return response.json()
        except json.JSONDecodeError:
            logger.error("La respuesta no es un JSON válido")
            raise ValueError("La respuesta no es un JSON válido")

    def request(
        self,
        method: str,
        endpoint: str,
        params: Optional[Dict[str, Any]] = None,
        data: Optional[Dict[str, Any]] = None,
        json_data: Optional[Dict[str, Any]] = None,
        headers: Optional[Dict[str, str]] = None
    ) -> Dict[str, Any]:
        """
        Realiza una petición HTTP segura a la API.

        Args:
            method: Método HTTP (GET, POST, PUT, DELETE, etc.)
            endpoint: Endpoint de la API (ej: 'users')
            params: Parámetros de query string
            data: Datos para enviar en el body (form-encoded)
            json_data: Datos para enviar como JSON en el body
            headers: Headers adicionales para la petición

        Returns:
            Diccionario con la respuesta JSON
        """
        url = urljoin(self.base_url, endpoint)

        # Validar que no haya inyección de parámetros maliciosos
        if params:
            for key, value in params.items():
                if not isinstance(key, str) or not isinstance(value, (str, int, float, bool)):
                    raise ValueError("Los parámetros deben ser strings o tipos básicos")

        # Configurar headers
        request_headers = self._get_headers()
        if headers:
            request_headers.update(headers)

        # Configurar SSL
        verify = self.ca_bundle_path if self.ca_bundle_path else self.verify_ssl

        try:
            response = self.session.request(
                method=method.upper(),
                url=url,
                params=params,
                data=data,
                json=json_data,
                headers=request_headers,
                timeout=self.timeout,
                verify=verify
            )
            return self._handle_response(response)
        except requests.exceptions.RequestException as e:
            logger.error(f"Error en la petición: {str(e)}")
            raise

    def get(self, endpoint: str, params: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
        """Realiza una petición GET."""
        return self.request("GET", endpoint, params=params)

    def post(self, endpoint: str, json_data: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
        """Realiza una petición POST."""
        return self.request("POST", endpoint, json_data=json_data)

    def put(self, endpoint: str, json_data: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
        """Realiza una petición PUT."""
        return self.request("PUT", endpoint, json_data=json_data)

    def delete(self, endpoint: str) -> Dict[str, Any]:
        """Realiza una petición DELETE."""
        return self.request("DELETE", endpoint)

    def set_token(self, token: str) -> None:
        """Establece o actualiza el token de autenticación."""
        self.token = token

# Ejemplo de uso
if __name__ == "__main__":
    try:
        # Configuración segura (el token debería venir de variables de entorno o un vault)
        API_BASE_URL = "https://api.ejemplo.com/v1/"
        API_TOKEN = os.getenv("API_TOKEN")  # Nunca hardcodear tokens en el código

        if not API_TOKEN:
            raise ValueError("El token de API no está configurado en las variables de entorno")

        # Crear cliente
        client = SecureAPIClient(
            base_url=API_BASE_URL,
            token=API_TOKEN,
            timeout=10,
            max_retries=3,
            verify_ssl=True
        )

        # Ejemplo de petición GET
        response = client.get("users", params={"limit": 10})
        print("Respuesta:", json.dumps(response, indent=2))

    except Exception as e:
        logger.error(f"Error en el ejemplo: {str(e)}")
```
