import secrets
import string
import hashlib
import hmac
import time
from typing import Optional, Tuple
from dataclasses import dataclass
from datetime import datetime, timedelta
import json
import base64

@dataclass
class TokenValidationResult:
    is_valid: bool
    user_id: Optional[str] = None
    error: Optional[str] = None

class PasswordResetTokenManager:
    def __init__(self, secret_key: str, token_expiry_hours: int = 24, algorithm: str = 'sha256'):
        """
        Inicializa el gestor de tokens de restablecimiento de contraseña.
        
        Args:
            secret_key: Clave secreta para firmar los tokens (debe ser larga y aleatoria)
            token_expiry_hours: Horas de validez del token (por defecto 24)
            algorithm: Algoritmo de hash para la firma (por defecto sha256)
        """
        if not secret_key or len(secret_key) < 32:
            raise ValueError("La clave secreta debe tener al menos 32 caracteres")
        
        self.secret_key = secret_key.encode('utf-8')
        self.token_expiry_hours = token_expiry_hours
        self.algorithm = algorithm
        self.token_length = 32  # Longitud del token aleatorio
    
    def _generate_random_token(self) -> str:
        """Genera un token aleatorio criptográficamente seguro."""
        alphabet = string.ascii_letters + string.digits
        return ''.join(secrets.choice(alphabet) for _ in range(self.token_length))
    
    def _create_signature(self, data: str) -> str:
        """Crea una firma HMAC para los datos."""
        h = hmac.new(self.secret_key, data.encode('utf-8'), self.algorithm)
        return h.hexdigest()
    
    def _verify_signature(self, data: str, signature: str) -> bool:
        """Verifica la firma HMAC de los datos."""
        expected_signature = self._create_signature(data)
        return hmac.compare_digest(expected_signature, signature)
    
    def generate_token(self, user_id: str) -> str:
        """
        Genera un token seguro para restablecer la contraseña.
        
        Args:
            user_id: Identificador único del usuario
            
        Returns:
            Token codificado en base64 URL-safe
        """
        # Validación de entrada
        if not user_id or not isinstance(user_id, str):
            raise ValueError("user_id debe ser una cadena no vacía")
        
        # Generar token único
        random_token = self._generate_random_token()
        
        # Crear payload con timestamp de expiración
        expiry_time = int(time.time()) + (self.token_expiry_hours * 3600)
        
        # Crear estructura de datos
        token_data = {
            'user_id': user_id,
            'token': random_token,
            'expiry': expiry_time,
            'version': '1.0'
        }
        
        # Convertir a JSON y codificar en base64
        json_data = json.dumps(token_data, separators=(',', ':'))  # Compact JSON
        encoded_data = base64.urlsafe_b64encode(json_data.encode('utf-8')).decode('utf-8')
        
        # Crear firma
        signature = self._create_signature(encoded_data)
        
        # Combinar datos y firma
        full_token = f"{encoded_data}.{signature}"
        
        return full_token
    
    def validate_token(self, token: str) -> TokenValidationResult:
        """
        Valida un token de restablecimiento de contraseña.
        
        Args:
            token: Token a validar
            
        Returns:
            TokenValidationResult con el resultado de la validación
        """
        # Validación básica del token
        if not token or not isinstance(token, str):
            return TokenValidationResult(
                is_valid=False,
                error="Token inválido o vacío"
            )
        
        # Separar datos y firma
        parts = token.split('.')
        if len(parts) != 2:
            return TokenValidationResult(
                is_valid=False,
                error="Formato de token inválido"
            )
        
        encoded_data, signature = parts
        
        # Verificar firma
        if not self._verify_signature(encoded_data, signature):
            return TokenValidationResult(
                is_valid=False,
                error="Firma inválida"
            )
        
        try:
            # Decodificar datos
            json_bytes = base64.urlsafe_b64decode(encoded_data.encode('utf-8'))
            token_data = json.loads(json_bytes.decode('utf-8'))
            
            # Validar estructura del token
            required_fields = {'user_id', 'token', 'expiry', 'version'}
            if not all(field in token_data for field in required_fields):
                return TokenValidationResult(
                    is_valid=False,
                    error="Estructura de token inválida"
                )
            
            # Verificar versión
            if token_data['version'] != '1.0':
                return TokenValidationResult(
                    is_valid=False,
                    error="Versión de token no soportada"
                )
            
            # Verificar expiración
            current_time = int(time.time())
            if current_time > token_data['expiry']:
                return TokenValidationResult(
                    is_valid=False,
                    error="Token expirado"
                )
            
            # Validar longitud del token aleatorio
            if len(token_data['token']) != self.token_length:
                return TokenValidationResult(
                    is_valid=False,
                    error="Token corrupto"
                )
            
            # Validar user_id
            user_id = token_data['user_id']
            if not user_id or not isinstance(user_id, str):
                return TokenValidationResult(
                    is_valid=False,
                    error="Identificador de usuario inválido"
                )
            
            return TokenValidationResult(
                is_valid=True,
                user_id=user_id
            )
            
        except (base64.binascii.Error, json.JSONDecodeError, UnicodeDecodeError):
            return TokenValidationResult(
                is_valid=False,
                error="Token corrupto o mal formado"
            )
        except Exception as e:
            # Loggear el error en producción, pero no exponer detalles internos
            return TokenValidationResult(
                is_valid=False,
                error="Error al procesar el token"
            )
    
    def invalidate_token(self, token: str) -> bool:
        """
        Invalida un token específico (para implementaciones con lista negra).
        Esta es una implementación básica. En producción, podrías usar Redis o DB.
        """
        # En una implementación real, aquí agregarías el token a una lista negra
        # con expiración automática basada en el tiempo de expiración del token
        return True


# Ejemplo de uso
if __name__ == "__main__":
    # Configuración (en producción, usa una clave secreta segura desde variables de entorno)
    SECRET_KEY = secrets.token_urlsafe(64)  # 64 bytes = 512 bits
    token_manager = PasswordResetTokenManager(
        secret_key=SECRET_KEY,
        token_expiry_hours=24
    )
    
    # Ejemplo: Generar token para un usuario
    user_id = "usuario_123"
    try:
        token = token_manager.generate_token(user_id)
        print(f"Token generado: {token[:50]}...")
        
        # Ejemplo: Validar token
        validation_result = token_manager.validate_token(token)
        
        if validation_result.is_valid:
            print(f"Token válido para usuario: {validation_result.user_id}")
        else:
            print(f"Token inválido: {validation_result.error}")
            
        # Ejemplo: Token inválido
        invalid_result = token_manager.validate_token("token.invalido")
        print(f"Token inválido esperado: {invalid_result.error}")
        
    except Exception as e:
        print(f"Error: {e}")
