```python
import os
from flask import Flask, request, jsonify, current_app
from werkzeug.utils import secure_filename
from werkzeug.security import safe_join
import imghdr
from functools import wraps
import uuid

app = Flask(__name__)

# Configuración segura
app.config.update(
    MAX_CONTENT_LENGTH=2 * 1024 * 1024,  # Limitar tamaño a 2MB
    UPLOAD_FOLDER='profile_images',
    ALLOWED_EXTENSIONS={'png', 'jpg', 'jpeg', 'gif'},
    ALLOWED_MIME_TYPES={'image/png', 'image/jpeg', 'image/gif'}
)

# Crear directorio de subidas si no existe
os.makedirs(app.config['UPLOAD_FOLDER'], exist_ok=True)

def validate_image(file_stream):
    """Validar que el archivo sea realmente una imagen"""
    try:
        image_type = imghdr.what(file_stream)
        if image_type not in app.config['ALLOWED_EXTENSIONS']:
            return False
        return True
    except:
        return False

def allowed_file(filename):
    """Verificar extensión del archivo"""
    return '.' in filename and \
           filename.rsplit('.', 1)[1].lower() in app.config['ALLOWED_EXTENSIONS']

def require_authentication(f):
    """Decorator para requerir autenticación (simplificado)"""
    @wraps(f)
    def decorated_function(*args, **kwargs):
        # En una aplicación real, aquí iría la verificación de autenticación
        # Por ejemplo: verificar token JWT o sesión
        if not request.headers.get('Authorization'):
            return jsonify({'error': 'Authentication required'}), 401
        return f(*args, **kwargs)
    return decorated_function

@app.route('/api/profile/image', methods=['POST'])
@require_authentication
def upload_profile_image():
    # Verificar que el request tenga el archivo
    if 'file' not in request.files:
        return jsonify({'error': 'No file part in the request'}), 400

    file = request.files['file']

    # Verificar que el usuario seleccionó un archivo
    if file.filename == '':
        return jsonify({'error': 'No file selected'}), 400

    # Verificar extensión y tipo MIME
    if not allowed_file(file.filename):
        return jsonify({'error': 'File extension not allowed'}), 400

    # Verificar tipo MIME real
    file.stream.seek(0)
    if file.content_type not in app.config['ALLOWED_MIME_TYPES']:
        return jsonify({'error': 'File type not allowed'}), 400

    # Verificar que es realmente una imagen
    file.stream.seek(0)
    if not validate_image(file.stream):
        return jsonify({'error': 'Invalid image file'}), 400

    # Generar nombre de archivo seguro y único
    filename = secure_filename(file.filename)
    unique_id = str(uuid.uuid4())
    file_ext = os.path.splitext(filename)[1]
    safe_filename = f"{unique_id}{file_ext}"

    # Verificar que el path resultante sea seguro
    try:
        save_path = safe_join(current_app.config['UPLOAD_FOLDER'], safe_filename)
        if not save_path:
            raise ValueError("Invalid path")
    except:
        return jsonify({'error': 'Invalid file path'}), 400

    # Guardar archivo
    try:
        file.stream.seek(0)
        file.save(save_path)
    except Exception as e:
        app.logger.error(f"Error saving file: {str(e)}")
        return jsonify({'error': 'Error saving file'}), 500

    # En una aplicación real, aquí guardarías la referencia en la base de datos
    # Por ejemplo: user.profile_image = safe_filename

    return jsonify({
        'message': 'File uploaded successfully',
        'filename': safe_filename
    }), 201

if __name__ == '__main__':
    # Configuración de producción (en desarrollo usar app.run())
    app.run(host='0.0.0.0', port=5000, threaded=False)
```
