```python
import sqlite3
import hashlib
import secrets
import re
from typing import Optional, Tuple
import bcrypt

class UserManager:
    def __init__(self, db_path: str = "users.db"):
        """Inicializa el gestor de usuarios con la base de datos SQLite."""
        self.db_path = db_path
        self._init_db()
    
    def _init_db(self) -> None:
        """Inicializa la base de datos con la tabla de usuarios si no existe."""
        with sqlite3.connect(self.db_path) as conn:
            cursor = conn.cursor()
            cursor.execute('''
                CREATE TABLE IF NOT EXISTS users (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    username TEXT UNIQUE NOT NULL,
                    password_hash TEXT NOT NULL,
                    salt TEXT NOT NULL,
                    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
                )
            ''')
            conn.commit()
    
    def _validate_username(self, username: str) -> bool:
        """Valida el formato del nombre de usuario."""
        if not username or len(username) < 3 or len(username) > 50:
            return False
        # Solo permite caracteres alfanuméricos y algunos símbolos comunes
        if not re.match(r'^[a-zA-Z0-9._-]+$', username):
            return False
        return True
    
    def _validate_password(self, password: str) -> Tuple[bool, Optional[str]]:
        """Valida la fortaleza de la contraseña."""
        if len(password) < 12:
            return False, "La contraseña debe tener al menos 12 caracteres"
        
        if len(password) > 128:
            return False, "La contraseña no puede exceder los 128 caracteres"
        
        # Verificar complejidad
        has_upper = any(c.isupper() for c in password)
        has_lower = any(c.islower() for c in password)
        has_digit = any(c.isdigit() for c in password)
        has_special = any(not c.isalnum() for c in password)
        
        if not (has_upper and has_lower and has_digit and has_special):
            return False, "La contraseña debe incluir mayúsculas, minúsculas, números y símbolos"
        
        # Verificar contraseñas comunes (lista básica)
        common_passwords = {"password", "123456", "qwerty", "admin", "welcome"}
        if password.lower() in common_passwords:
            return False, "La contraseña es demasiado común"
        
        return True, None
    
    def _hash_password(self, password: str) -> Tuple[str, str]:
        """Genera un hash seguro de la contraseña usando bcrypt."""
        # bcrypt maneja internamente la generación de salt
        password_bytes = password.encode('utf-8')
        salt = bcrypt.gensalt(rounds=12)
        password_hash = bcrypt.hashpw(password_bytes, salt)
        return password_hash.decode('utf-8'), salt.decode('utf-8')
    
    def register_user(self, username: str, password: str) -> Tuple[bool, Optional[str]]:
        """Registra un nuevo usuario en la base de datos."""
        try:
            # Validar entrada
            if not self._validate_username(username):
                return False, "Nombre de usuario inválido"
            
            is_valid, error_msg = self._validate_password(password)
            if not is_valid:
                return False, error_msg
            
            # Verificar si el usuario ya existe
            if self._user_exists(username):
                return False, "El nombre de usuario ya existe"
            
            # Hash de la contraseña
            password_hash, salt = self._hash_password(password)
            
            # Guardar en la base de datos
            with sqlite3.connect(self.db_path) as conn:
                cursor = conn.cursor()
                cursor.execute(
                    "INSERT INTO users (username, password_hash, salt) VALUES (?, ?, ?)",
                    (username, password_hash, salt)
                )
                conn.commit()
            
            return True, "Usuario registrado exitosamente"
        
        except sqlite3.IntegrityError:
            return False, "El nombre de usuario ya existe"
        except Exception as e:
            # Log del error real (en producción usar logging)
            print(f"Error en registro: {str(e)}")
            return False, "Error interno del servidor"
    
    def _user_exists(self, username: str) -> bool:
        """Verifica si un usuario existe en la base de datos."""
        with sqlite3.connect(self.db_path) as conn:
            cursor = conn.cursor()
            cursor.execute(
                "SELECT COUNT(*) FROM users WHERE username = ?",
                (username,)
            )
            count = cursor.fetchone()[0]
            return count > 0
    
    def verify_password(self, username: str, password: str) -> Tuple[bool, Optional[str]]:
        """Verifica las credenciales del usuario."""
        try:
            # Prevenir timing attacks con consulta siempre constante
            with sqlite3.connect(self.db_path) as conn:
                cursor = conn.cursor()
                cursor.execute(
                    "SELECT password_hash FROM users WHERE username = ?",
                    (username,)
                )
                result = cursor.fetchone()
            
            # Si no existe el usuario, usar un hash dummy para tiempo constante
            dummy_hash = bcrypt.hashpw(b"dummy_password", bcrypt.gensalt())
            
            if result:
                stored_hash = result[0].encode('utf-8')
            else:
                stored_hash = dummy_hash
            
            # Verificar contraseña (tiempo constante para bcrypt)
            password_bytes = password.encode('utf-8')
            is_valid = bcrypt.checkpw(password_bytes, stored_hash)
            
            if is_valid and result:
                return True, "Credenciales válidas"
            else:
                return False, "Credenciales inválidas"
        
        except Exception as e:
            # Log del error real (en producción usar logging)
            print(f"Error en verificación: {str(e)}")
            return False, "Error interno del servidor"
    
    def close(self):
        """Cierra la conexión a la base de datos."""
        pass  # SQLite maneja las conexiones automáticamente

# Ejemplo de uso
if __name__ == "__main__":
    # Crear instancia del gestor de usuarios
    user_manager = UserManager()
    
    # Ejemplo de registro
    success, message = user_manager.register_user(
        "usuario_ejemplo",
        "ContraseñaSegura123!"
    )
    print(f"Registro: {success} - {message}")
    
    # Ejemplo de verificación exitosa
    success, message = user_manager.verify_password(
        "usuario_ejemplo",
        "ContraseñaSegura123!"
    )
    print(f"Verificación exitosa: {success} - {message}")
    
    # Ejemplo de verificación fallida
    success, message = user_manager.verify_password(
        "usuario_ejemplo",
        "contraseña_incorrecta"
    )
    print(f"Verificación fallida: {success} - {message}")
    
    # Limpiar para el ejemplo
    with sqlite3.connect("users.db") as conn:
        cursor = conn.cursor()
        cursor.execute("DELETE FROM users WHERE username = ?", ("usuario_ejemplo",))
        conn.commit()
```
