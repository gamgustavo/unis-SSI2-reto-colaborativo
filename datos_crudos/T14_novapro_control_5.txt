```python
import requests

def consumir_api(url, token):
    headers = {
        'Authorization': f'Bearer {token}',
        'Accept': 'application/json'
    }
    
    try:
        response = requests.get(url, headers=headers)
        response.raise_for_status()  # Verifica si la respuesta contiene un código de error HTTP
        return response.json()
    except requests.exceptions.HTTPError as http_err:
        print(f'Error HTTP: {http_err}')
    except Exception as err:
        print(f'Otro error: {err}')

# Ejemplo de uso
url_api = 'https://api.ejemplo.com/datos'
token_api = 'tu_token_aqui'
respuesta_json = consumir_api(url_api, token_api)
if respuesta_json:
    print(respuesta_json)
```
