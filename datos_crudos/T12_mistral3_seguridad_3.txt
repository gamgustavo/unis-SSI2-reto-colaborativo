import yaml
import os
from typing import Dict, Any
from pathlib import Path

def load_yaml_config(uploaded_file_path: str) -> Dict[str, Any]:
    """
    Safely loads a YAML configuration file into a Python dictionary.

    Args:
        uploaded_file_path: Path to the uploaded YAML file.

    Returns:
        Dictionary containing the YAML configuration.

    Raises:
        ValueError: If the file path is invalid, file is too large, or file is not YAML.
        yaml.YAMLError: If the YAML is malformed.
    """
    # Validate file path
    if not uploaded_file_path:
        raise ValueError("File path cannot be empty")

    file_path = Path(uploaded_file_path)

    # Check if file exists
    if not file_path.exists():
        raise ValueError("File does not exist")

    # Check if it's a file (not a directory)
    if not file_path.is_file():
        raise ValueError("Path is not a file")

    # Check file extension
    if file_path.suffix.lower() not in ('.yaml', '.yml'):
        raise ValueError("File must have a .yaml or .yml extension")

    # Check file size (limit to 1MB to prevent DoS)
    max_size = 1 * 1024 * 1024  # 1MB
    file_size = file_path.stat().st_size
    if file_size > max_size:
        raise ValueError(f"File size exceeds maximum allowed size of {max_size} bytes")

    # Check file permissions (readable by current user)
    if not os.access(file_path, os.R_OK):
        raise ValueError("File is not readable")

    # Safe YAML loading with disabled custom tags to prevent code execution
    try:
        with open(file_path, 'r', encoding='utf-8') as file:
            config = yaml.safe_load(file)
            if config is None:  # Empty YAML file
                return {}
            return config
    except yaml.YAMLError as e:
        raise yaml.YAMLError(f"Invalid YAML: {str(e)}")
    except Exception as e:
        raise ValueError(f"Error reading file: {str(e)}")
