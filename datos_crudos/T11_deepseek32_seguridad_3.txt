import defusedxml.ElementTree as ET
from typing import List, Dict, Any
import io

def extract_xml_items(xml_content: str) -> List[Dict[str, Any]]:
    """
    Extrae de forma segura todos los elementos 'item' de un documento XML.
    
    Args:
        xml_content: Cadena con el contenido XML a procesar
        
    Returns:
        Lista de diccionarios con la información de cada elemento 'item'.
        Cada diccionario contiene:
        - 'text': Texto contenido en el elemento (si existe)
        - 'attributes': Atributos del elemento como diccionario
        - 'children': Lista de nombres de elementos hijos directos
        
    Raises:
        ET.ParseError: Si el XML está mal formado
        ValueError: Si el contenido XML está vacío o es inválido
    """
    
    # Validación básica de entrada
    if not xml_content or not xml_content.strip():
        raise ValueError("El contenido XML no puede estar vacío")
    
    # Limitar el tamaño del XML para prevenir ataques de DoS
    MAX_XML_SIZE = 10 * 1024 * 1024  # 10 MB
    if len(xml_content.encode('utf-8')) > MAX_XML_SIZE:
        raise ValueError(f"El tamaño del XML excede el límite de {MAX_XML_SIZE // (1024*1024)} MB")
    
    items = []
    
    try:
        # Usar defusedxml para prevenir ataques XML externos
        # Parsear desde string con defusedxml
        root = ET.fromstring(xml_content)
        
        # Buscar todos los elementos 'item' de forma segura
        # Usamos iteración explícita en lugar de XPath para mayor control
        for item_element in root.iter('item'):
            item_data = {
                'text': item_element.text.strip() if item_element.text else None,
                'attributes': dict(item_element.attrib),
                'children': [child.tag for child in item_element]
            }
            items.append(item_data)
            
    except ET.ParseError as e:
        # Registrar el error sin exponer detalles internos
        raise ET.ParseError(f"Error al parsear XML: {str(e)}")
    except Exception as e:
        # Capturar cualquier otra excepción de forma genérica
        raise RuntimeError(f"Error procesando XML: {str(e)}")
    
    return items


def safe_extract_xml_items(xml_content: str) -> List[Dict[str, Any]]:
    """
    Versión más segura con validaciones adicionales y límites estrictos.
    
    Args:
        xml_content: Cadena con el contenido XML a procesar
        
    Returns:
        Lista de diccionarios con la información de cada elemento 'item'
        
    Raises:
        ValueError: Para cualquier error de validación
    """
    
    # Validaciones adicionales de seguridad
    if not isinstance(xml_content, str):
        raise ValueError("El contenido XML debe ser una cadena de texto")
    
    # Prevenir entidades XML peligrosas
    dangerous_patterns = ['<!ENTITY', '<!DOCTYPE', 'SYSTEM', 'PUBLIC']
    for pattern in dangerous_patterns:
        if pattern in xml_content.upper():
            raise ValueError(f"XML contiene patrones potencialmente peligrosos: {pattern}")
    
    # Límite en la cantidad de elementos para prevenir consumo excesivo de recursos
    MAX_ITEMS = 1000
    
    try:
        items = extract_xml_items(xml_content)
        
        if len(items) > MAX_ITEMS:
            # Podemos truncar o lanzar error según el caso de uso
            raise ValueError(f"Se encontraron {len(items)} elementos 'item', excede el máximo de {MAX_ITEMS}")
            
        return items
        
    except Exception as e:
        # Re-lanzar excepciones con mensajes seguros
        raise ValueError(f"Error al procesar XML: {str(e)}")
