from flask import Flask, request, jsonify
import sqlite3
import re

app = Flask(__name__)

def get_db_connection():
    """Establece conexión segura con la base de datos"""
    conn = sqlite3.connect('database.db')
    conn.row_factory = sqlite3.Row
    return conn

def validate_input(nombre):
    """Valida que el nombre cumpla con un patrón seguro"""
    # Permitir solo letras, números, espacios y algunos caracteres comunes
    # Ajustar según los requisitos específicos de la aplicación
    if not nombre or len(nombre) > 100:
        return False
    
    # Patrón seguro: letras, números, espacios y caracteres básicos
    pattern = r'^[a-zA-Z0-9áéíóúÁÉÍÓÚñÑ\s\-_\.]+$'
    return bool(re.match(pattern, nombre))

@app.route('/buscar_usuario', methods=['GET'])
def buscar_usuario():
    """Endpoint seguro para buscar usuario por nombre"""
    
    # 1. Validar que el parámetro 'nombre' existe
    nombre = request.args.get('nombre')
    if not nombre:
        return jsonify({'error': 'Parámetro "nombre" requerido'}), 400
    
    # 2. Validar formato del input
    if not validate_input(nombre):
        return jsonify({'error': 'Nombre con formato inválido'}), 400
    
    # 3. Limitar longitud del parámetro
    if len(nombre) > 100:
        return jsonify({'error': 'Nombre demasiado largo'}), 400
    
    conn = None
    try:
        # 4. Usar consultas parametrizadas para prevenir SQL injection
        conn = get_db_connection()
        cursor = conn.cursor()
        
        # Consulta segura con parámetros
        cursor.execute(
            "SELECT id, nombre, email, fecha_creacion FROM usuarios WHERE nombre = ? LIMIT 1",
            (nombre,)
        )
        
        usuario = cursor.fetchone()
        
        if usuario:
            # 5. Convertir resultado a diccionario de forma segura
            datos_usuario = dict(usuario)
            
            # 6. Sanitizar datos sensibles si es necesario
            # (En este caso solo mostramos datos básicos)
            
            return jsonify({
                'success': True,
                'usuario': datos_usuario
            }), 200
        else:
            # 7. No revelar información sensible al usuario no encontrado
            return jsonify({
                'success': False,
                'message': 'Usuario no encontrado'
            }), 404
            
    except sqlite3.Error as e:
        # 8. Loggear el error internamente (en producción usar logging)
        # logging.error(f"Error de base de datos: {str(e)}")
        return jsonify({
            'error': 'Error interno del servidor'
        }), 500
        
    except Exception as e:
        # 9. Manejo genérico de excepciones
        # logging.error(f"Error inesperado: {str(e)}")
        return jsonify({
            'error': 'Error interno del servidor'
        }), 500
        
    finally:
        # 10. Asegurar cierre de conexión
        if conn:
            conn.close()

# Configuración de seguridad adicional
@app.after_request
def add_security_headers(response):
    """Añade headers de seguridad HTTP"""
    response.headers['X-Content-Type-Options'] = 'nosniff'
    response.headers['X-Frame-Options'] = 'DENY'
    response.headers['X-XSS-Protection'] = '1; mode=block'
    response.headers['Content-Security-Policy'] = "default-src 'self'"
    return response

if __name__ == '__main__':
    # Configuración segura para producción
    app.run(
        debug=False,  # Desactivar debug en producción
        host='0.0.0.0',
        port=5000
    )
